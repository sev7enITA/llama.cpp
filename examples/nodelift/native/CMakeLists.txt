cmake_minimum_required(VERSION 3.22)
project(nodelift LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/llama.cpp" CACHE PATH "Pinned llama.cpp checkout")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
if(ANDROID)
  # Debug signing/UI diagnostics must not silently disable inference optimization.
  # Keep symbols and assertions, and preserve the baseline ARM ISA. This applies
  # to llama/ggml subdirectories as well as our wrapper; no fast-math is enabled.
  option(NODELIFT_OPTIMIZE_ANDROID_DEBUG "Optimize native Android debug kernels" ON)
  if(NODELIFT_OPTIMIZE_ANDROID_DEBUG)
    add_compile_options("$<$<CONFIG:Debug>:-O3>")
  endif()
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
  set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
  set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
option(NODELIFT_EXPERIMENTAL_DEFERRED_DRAFT "Experimental draft KV optimization, not qualified for production" OFF)
add_library(nodelift SHARED core.cpp "${LLAMA_SOURCE_DIR}/vendor/hash/sha256/sha256.c")
if(NODELIFT_EXPERIMENTAL_DEFERRED_DRAFT)
  target_compile_definitions(nodelift PRIVATE NODELIFT_EXPERIMENTAL_DEFERRED_DRAFT=1)
endif()
target_include_directories(nodelift PRIVATE "${LLAMA_SOURCE_DIR}/vendor" "${LLAMA_SOURCE_DIR}/vendor/hash" "${LLAMA_SOURCE_DIR}/vendor/hash/sha256")
target_link_libraries(nodelift PRIVATE llama)
if(ANDROID)
  target_sources(nodelift PRIVATE jni.cpp)
  target_link_libraries(nodelift PRIVATE log)
endif()
set_target_properties(nodelift PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set_target_properties(nodelift PROPERTIES FRAMEWORK TRUE
    FRAMEWORK_VERSION A MACOSX_FRAMEWORK_IDENTIFIER dev.nodelift.engine
    PUBLIC_HEADER core.h MACOSX_FRAMEWORK_SHORT_VERSION_STRING 0.2.1
    MACOSX_FRAMEWORK_BUNDLE_VERSION 3)
endif()

option(NODELIFT_BUILD_SHAPE_PROBE "Build experimental batch shape numerical probe" OFF)
if(NODELIFT_BUILD_SHAPE_PROBE AND NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(nodelift-shape-probe experiments/shape_probe.cpp)
  target_include_directories(nodelift-shape-probe PRIVATE "${LLAMA_SOURCE_DIR}/vendor")
  target_link_libraries(nodelift-shape-probe PRIVATE llama)
endif()
